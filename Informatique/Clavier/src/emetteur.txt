#include <Arduino.h>
#include <esp_now.h>
#include <WiFi.h>

// REMPLACEZ AVEC L'ADRESSE MAC DU RÉCEPTEUR
uint8_t broadcastAddress[] = {0x3C,0xDC,0x75,0x63,0x62,0xE0}; 

// Structure du message
typedef struct struct_message {
  char texte;
} struct_message;

struct_message monMessage;
esp_now_peer_info_t peerInfo;

// Callback quand les données sont envoyées
void OnDataSent(const uint8_t *mac_addr, esp_now_send_status_t status) {
  Serial.print("\r\nStatut du dernier envoi : ");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "Succès" : "Échec");
}

void setup() {
  Serial.begin(115200);
  WiFi.mode(WIFI_STA);

  if (esp_now_init() != ESP_OK) {
    Serial.println("Erreur initialisation ESP-NOW");
    return;
  }

  esp_now_register_send_cb(OnDataSent);
  
  // Enregistrer le pair (peer)
  memcpy(peerInfo.peer_addr, broadcastAddress, 6);
  peerInfo.channel = 0;  
  peerInfo.encrypt = false;
  
  if (esp_now_add_peer(&peerInfo) != ESP_OK){
    Serial.println("Échec de l'ajout du pair");
    return;
  }
}

void loop() {
  if(Serial.available() > 0) {
    monMessage.texte = Serial.read(); // Par exemple, envoyer 'S' à 
    esp_err_t result = esp_now_send(broadcastAddress, (uint8_t *) &monMessage, sizeof(monMessage));
    const char* status = (result == ESP_OK) ? "Message envoyé avec succès" : "Erreur d'envoi";
    Serial.println(status);
  }
}
